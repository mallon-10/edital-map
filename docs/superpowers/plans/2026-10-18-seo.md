  { slug: 'veiculos', nome: 'veículos e peças', termo: 'veículo', grupo: 'veículos e peças' },
  { slug: 'combustiveis', nome: 'combustíveis', termo: 'combustível', grupo: 'combustíveis' },

  { slug: 'manutencao-predial', nome: 'manutenção predial', termo: 'manutenção predial', grupo: 'manutenção predial' },
  { slug: 'servicos-de-saude', nome: 'serviços de saúde', termo: 'serviço de saúde', grupo: 'serviços de saúde' },
];

const POR_SLUG = new Map(ASSUNTOS.map((a) => [a.slug, a]));
const UFS = new Set(UFS_BRASIL);

export function assuntoPorSlug(slug: string): Assunto | null {
  return POR_SLUG.get(slug) ?? null;
}

/** 'sp' → 'SP'. Só minúsculas: '/licitacoes/SP' é 404 (um endereço por página). */
export function ufPorSlug(slug: string): string | null {
  if (slug !== slug.toLowerCase()) return null;
  const uf = slug.toUpperCase();
  return UFS.has(uf) ? uf : null;
}

/** Segmento de /licitacoes/[x]: UF tem 2 letras, assunto nunca (teste garante). */
export function resolverSegmento(x: string): { tipo: 'assunto'; assunto: Assunto } | { tipo: 'uf'; uf: string } | null {
  const assunto = assuntoPorSlug(x);
  if (assunto) return { tipo: 'assunto', assunto };
  const uf = ufPorSlug(x);
  return uf ? { tipo: 'uf', uf } : null;
}

const NA = new Set(['BA', 'PB']);
const EM = new Set(['AL', 'GO', 'MG', 'PE', 'RO', 'RR', 'SC', 'SP', 'SE']);

/** "em São Paulo", "na Bahia", "no Rio de Janeiro". */
export function naUf(uf: string): string {
  const prep = NA.has(uf) ? 'na' : EM.has(uf) ? 'em' : 'no';
  return `${prep} ${NOMES_UF[uf]}`;
}

```ts
import { NOMES_UF } from '../estados';
import { data, moeda, numero } from '../formato';
import { modalidadeCurta } from '../participacao';
import { nomeProprio } from '../texto';
import type { Edital } from '../tipos';
import { type Assunto, naUf } from './assuntos';
import { url } from './site';

/** Uma página de /licitacoes: Brasil (nada), só UF, só assunto ou assunto na UF. */
export type Pagina = { assunto: Assunto | null; uf: string | null };

export function caminhoDaPagina(p: Pagina): string {
  const partes = ['/licitacoes'];
  if (p.assunto) partes.push(p.assunto.slug);
  if (p.uf) partes.push(p.uf.toLowerCase());
  return partes.join('/');
}

const onde = (p: Pagina) => (p.uf ? naUf(p.uf) : 'no Brasil');
const de = (p: Pagina) => (p.assunto ? ` de ${p.assunto.nome}` : '');

export function tituloDaPagina(p: Pagina): string {
  return `Licitações${de(p)} abertas ${onde(p)}`;
}

/** "1.234 editais de EPI abertos na Bahia" / "1 edital aberto em São Paulo" / "Nenhum edital de EPI aberto no Brasil agora". */
function contagem(p: Pagina, total: number): string {
  if (total === 0) return `Nenhum edital${de(p)} aberto ${onde(p)} agora`;
  return total === 1 ? `1 edital${de(p)} aberto ${onde(p)}` : `${numero(total)} editais${de(p)} abertos ${onde(p)}`;
}

export function descricaoDaPagina(p: Pagina, total: number): string {
  if (total === 0) return `${contagem(p, 0)}. O Edital Map mostra as licitações do PNCP no mapa, atualizadas a cada 3 horas.`;
  return `${contagem(p, total)}, direto do PNCP e atualizados a cada 3 horas. Veja prazos, valores e órgãos no mapa.`;
}

export function introDaPagina(p: Pagina, total: number): string {
  const busca = p.assunto ? ` A busca inclui termos parecidos com "${p.assunto.termo}", no objeto e nos itens do edital.` : '';
  if (total === 0) return `${contagem(p, 0)}. Os editais chegam do PNCP (Portal Nacional de Contratações Públicas) a cada 3 horas.${busca}`;
  return `${contagem(p, total)} no PNCP (Portal Nacional de Contratações Públicas), com prazo para enviar proposta. A lista é atualizada a cada 3 horas e mostra primeiro os que encerram antes.${busca}`;
}

export function buscaDaPagina(p: Pagina): string {
  const q = new URLSearchParams();
  if (p.assunto) q.set('q', p.assunto.termo);
  if (p.uf) q.set('uf', p.uf);
  const t = q.toString();
  return t ? `/?${t}` : '/';
}

export function migalhas(p: Pagina): Array<{ nome: string; caminho: string }> {
  const itens = [{ nome: 'Início', caminho: '/' }, { nome: 'Licitações', caminho: '/licitacoes' }];
  if (p.assunto) itens.push({ nome: p.assunto.nome.replace(/^./, (c) => c.toUpperCase()), caminho: caminhoDaPagina({ assunto: p.assunto, uf: null }) });
  if (p.uf) itens.push({ nome: NOMES_UF[p.uf], caminho: caminhoDaPagina(p) });
  return itens;
}

export function jsonLdMigalhas(itens: Array<{ nome: string; caminho: string }>): object {
  return {
    '@context': 'https://schema.org',
    '@type': 'BreadcrumbList',
    itemListElement: itens.map((i, n) => ({ '@type': 'ListItem', position: n + 1, name: i.nome, item: url(i.caminho) })),
  };
}

/** Mesma regra de public.em_aberto (0011): prazo no futuro e situação diferente de Revogada (2) e Anulada (3). */
export function editalAberto(e: { encerramento_em: string | null; situacao_id: number | null }, agora: Date): boolean {
  if (!e.encerramento_em || Date.parse(e.encerramento_em) <= agora.getTime()) return false;
  return e.situacao_id !== 2 && e.situacao_id !== 3;
}

const MAX_DESCRICAO = 160;

export function descricaoDoEdital(e: Edital): string {
  const partes = [
    nomeProprio(e.orgao_nome),
    `${e.municipio_nome ?? '—'}/${e.uf}`,
    e.modalidade_nome ? modalidadeCurta(e.modalidade_nome) : null,
    e.encerramento_em ? `propostas até ${data(e.encerramento_em)}` : null,
    moeda(e.valor_estimado) ?? 'Valor não informado',
  ].filter(Boolean);
  const t = partes.join(' · ');
  return t.length <= MAX_DESCRICAO ? t : `${t.slice(0, MAX_DESCRICAO - 1)}…`;
}

/** Parâmetros que formam uma busca na página inicial (lib/urlBusca.ts); utm_*, gclid e afins não contam. */
const PARAMETROS_BUSCA = ['q', 'uf', 'municipio', 'valor', 'modalidades', 'orgao', 'raio', 'dias', 'ordem', 'pagina', 'vista', 'tela'];

export function temBusca(params: Record<string, string | string[] | undefined>): boolean {
  return PARAMETROS_BUSCA.some((k) => params[k] !== undefined);
}

/** Editais abertos por UF para um termo de assunto (com os relacionados, como a busca); sem termo, todos. */
export function abertosPorUf(q: string | null): Promise<Record<string, number>> {
  return cache(`uf:${q ?? ''}`, async () => {
    const b = interpretarBusca(q);
    const { data, error } = await supabaseServidor().rpc('abertos_por_uf', {
      p_termos: b.digitados,
      p_relacionados: b.relacionados.map((r) => r.termo),
      p_excluir: b.excluidos,
    });
    if (error) throw new Error(`abertos_por_uf: ${error.message}`);
    return Object.fromEntries(((data ?? []) as Linha[]).map((l) => [String(l.uf), Number(l.total)]));
  });
}

export function totalEditaisAbertos(): Promise<number> {
  return cache('total', async () => {
    const { data, error } = await supabaseServidor().rpc('total_editais_abertos', {});
    if (error) throw new Error(`total_editais_abertos: ${error.message}`);
    return Number(data ?? 0);
  });
}

export function editaisAbertosSitemap(limite: number, deslocamento: number): Promise<Array<{ slug: string; atualizada_pncp_em: string }>> {
  return cache(`sitemap:${limite}:${deslocamento}`, async () => {
    const { data, error } = await supabaseServidor().rpc('editais_abertos_sitemap', { p_limite: limite, p_deslocamento: deslocamento });
    if (error) throw new Error(`editais_abertos_sitemap: ${error.message}`);
    return ((data ?? []) as Linha[]).map((l) => ({ slug: String(l.slug), atualizada_pncp_em: String(l.atualizada_pncp_em) }));
  });
}
```

/** Links da página para as vizinhas. `porUf` (contagem do assunto por UF) esconde as UFs vazias; null = mostra todas. */
export function vizinhos(p: Pagina, porUf: Record<string, number> | null): Array<{ nome: string; caminho: string; total: number | null }> {
  const assuntos = (uf: string | null) =>
    ASSUNTOS.map((a) => ({ nome: maiuscula(a.nome), caminho: caminhoDaPagina({ assunto: a, uf }), total: null }));
  const ufs = (assunto: Pagina['assunto']) =>
    ufsPorNome()
      .filter((uf) => !porUf || (porUf[uf] ?? 0) > 0)
      .map((uf) => ({ nome: NOMES_UF[uf], caminho: caminhoDaPagina({ assunto, uf }), total: porUf ? porUf[uf] : null }));
  if (p.assunto && p.uf) {
    return [
      { nome: `${maiuscula(p.assunto.nome)} no Brasil`, caminho: caminhoDaPagina({ assunto: p.assunto, uf: null }), total: null },
      { nome: `Tudo ${naUf(p.uf)}`, caminho: caminhoDaPagina({ assunto: null, uf: p.uf }), total: null },
    ];
  }
  if (p.assunto) return ufs(p.assunto);
  if (p.uf) return assuntos(p.uf);
  return [...assuntos(null), ...ufs(null)];
}

import type { Assunto } from './assuntos';
import { caminhoDaPagina } from './textos';

/** Editais por arquivo do sitemap (o formato aceita até 50.000 endereços). */
export const POR_ARQUIVO = 40_000;

export function idsDoSitemap(totalEditais: number): Array<{ id: number }> {
  const arquivos = Math.ceil(totalEditais / POR_ARQUIVO);
  return Array.from({ length: arquivos + 1 }, (_, id) => ({ id }));
}

/** Caminhos de /licitacoes com pelo menos um edital aberto (contagem do assunto por UF; assunto null = todos). */
export function paginasFixas(contagens: Array<{ assunto: Assunto | null; porUf: Record<string, number> }>): string[] {
  const caminhos: string[] = [];
  for (const { assunto, porUf } of contagens) {
    const ufs = Object.keys(porUf).filter((uf) => porUf[uf] > 0).sort();
    if (assunto === null) caminhos.push('/licitacoes');
    else if (ufs.length === 0) continue;
    else caminhos.push(caminhoDaPagina({ assunto, uf: null }));
    for (const uf of ufs) caminhos.push(caminhoDaPagina({ assunto, uf }));
  }
  return caminhos;
}
