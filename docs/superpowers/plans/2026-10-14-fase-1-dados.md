# Edital Map · Fase 1 · Dados — plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pôr no Supabase um espelho fiel das contratações do PNCP de SC, PR e RS, com os itens das compras abertas, atualizado a cada 2 h pelo GitHub Actions, e provar com uma conferência automática que os dados batem com o PNCP.

**Architecture:** Um worker Node/TypeScript em `worker/` com unidades pequenas: cliente HTTP do PNCP (paginação, espera em 429/5xx), validação zod, mapeamento puro para linhas do banco, acesso ao Supabase via `supabase-js` (service role) e uma camada de sincronização que recebe as dependências por parâmetro (testável com fakes). Carga inicial usa `/publicacao` em janelas de 7 dias; a sincronização usa `/atualizacao` de ontem até hoje e só busca itens de compras abertas novas ou alteradas.

**Tech Stack:** Node 24, TypeScript 5, tsx 4, vitest 3, @supabase/supabase-js 2, zod 3, date-fns 4 + date-fns-tz 3, GitHub Actions, Supabase (Postgres 17, `sa-east-1`).

**Spec:** `docs/superpowers/specs/2026-10-14-fase-1-dados-design.md`

export const PARALELISMO = 3;
export const TENTATIVAS = 4;
export const TIMEOUT_MS = 30_000;
export const PAGINA_CONTRATACOES = 50;
export const PAGINA_ITENS = 50;
export const LOTE_BANCO = 500;
export const LIMITE_CATCH_UP = 300;
export const LIMITE_INVALIDOS = 0.05;
export const BASE_CONSULTA = 'https://pncp.gov.br/api/consulta';
export const BASE_PNCP = 'https://pncp.gov.br/api/pncp';

export const UFS_BRASIL = [
  'AC', 'AL', 'AM', 'AP', 'BA', 'CE', 'DF', 'ES', 'GO', 'MA', 'MG', 'MS', 'MT', 'PA',
  'PB', 'PE', 'PI', 'PR', 'RJ', 'RN', 'RO', 'RR', 'RS', 'SC', 'SE', 'SP', 'TO',
] as const;

export function lerUfs(valor: string | undefined): string[] {
  const texto = valor?.trim() ? valor : 'SC,PR,RS';
  const ufs = texto
    .split(',')
    .map((s) => s.trim().toUpperCase())
    .filter(Boolean);
  for (const uf of ufs) {
    if (!(UFS_BRASIL as readonly string[]).includes(uf)) throw new Error(`UF inválida: ${uf}`);
  }
  return [...new Set(ufs)];
}

export function lerModalidades(valor: string | undefined): number[] {
  if (!valor?.trim()) return [...MODALIDADES];
  const modalidades = valor.split(',').map((s) => Number(s.trim()));
  for (const m of modalidades) {
    if (!(MODALIDADES as readonly number[]).includes(m)) throw new Error(`Modalidade fora da v1: ${m}`);
  }
  return modalidades;
}

export function lerSupabase(env: NodeJS.ProcessEnv = process.env): { url: string; chave: string } {
  const url = env.SUPABASE_URL;
  const chave = env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !chave) {
    throw new Error('Defina SUPABASE_URL e SUPABASE_SERVICE_ROLE_KEY (arquivo .env ou Secrets do GitHub).');
  }
  return { url, chave };
}

import { formatInTimeZone, fromZonedTime } from 'date-fns-tz';

export const FUSO = 'America/Sao_Paulo';

/** Data do PNCP sem fuso ("2026-09-22T08:00:00"), lida como horário de Brasília, em ISO UTC. */
export function deBrasilia(valor: string | null | undefined): string | null {
  if (!valor || !valor.trim()) return null;
  const data = fromZonedTime(valor, FUSO);
  if (Number.isNaN(data.getTime())) throw new Error(`Data inválida do PNCP: ${valor}`);
  return data.toISOString();
}

export function hojeBrasilia(agora: Date): string {
  return formatInTimeZone(agora, FUSO, 'yyyy-MM-dd');
}

export function somarDias(dia: string, n: number): string {
  const data = new Date(`${dia}T00:00:00Z`);
  data.setUTCDate(data.getUTCDate() + n);
  return data.toISOString().slice(0, 10);
}

export function paraPncp(dia: string): string {
  return dia.replaceAll('-', '');
}

export function inicioDoDia(dia: string): string {
  return deBrasilia(`${dia}T00:00:00`) as string;
}
```

  return erro.issues.map((i) => `${i.path.join('.') || '(raiz)'}: ${i.message}`).join('; ');

export function validarContratacao(bruto: unknown): Validado<ContratacaoPncp> {

  const id = (bruto as { numeroControlePNCP?: unknown } | null)?.numeroControlePNCP;
  return {
    ok: false,
    id: typeof id === 'string' && id ? id : '(sem numeroControlePNCP)',

  };

  return { ok: false, id: `${contratacaoId}#${String(numero ?? '?')}`, motivo: motivo(r.error) };

    urls.push(String(url));

    if (proxima instanceof Error) throw proxima;

import {
  BASE_CONSULTA,
  BASE_PNCP,
  PAGINA_CONTRATACOES,
  PAGINA_ITENS,
  PARALELISMO,
  TENTATIVAS,
  TIMEOUT_MS,
} from '../config';
import { paraPncp } from '../datas';
import type { ChaveCompra } from '../mapear';

export class ErroPncp extends Error {
  constructor(
    mensagem: string,
    readonly status?: number,
    readonly url?: string,
  ) {
    super(mensagem);
    this.name = 'ErroPncp';
  }
}

export type FiltroContratacoes = { uf: string; modalidade: number; dataInicial: string; dataFinal: string };
export type PaginaContratacoes = { pagina: number; totalPaginas: number; totalRegistros: number; registros: unknown[] };
export type OpcoesCliente = {
  fetch?: typeof fetch;
  esperar?: (ms: number) => Promise<void>;
  aleatorio?: () => number;
  paralelismo?: number;
  tentativas?: number;
  timeoutMs?: number;
};
type RespostaContratacoes = { data?: unknown[]; totalRegistros?: number; totalPaginas?: number };

export function criarLimitador(maximo: number) {
  let ativos = 0;
  const fila: Array<() => void> = [];
  return async function limitar<T>(tarefa: () => Promise<T>): Promise<T> {
    if (ativos < maximo) ativos++;
    else await new Promise<void>((liberar) => fila.push(liberar));
    try {
      return await tarefa();
    } finally {
      // Passa a vaga direto para quem está esperando; só libera se a fila estiver vazia.
      const proxima = fila.shift();
      if (proxima) proxima();
      else ativos--;
    }
  };
}

function podeTentarDeNovo(status: number): boolean {
  return status === 429 || status >= 500;
}

function urlContratacoes(
  endpoint: 'publicacao' | 'atualizacao',
  filtro: FiltroContratacoes,
  pagina: number,
  tamanho: number,
): string {
  const params = new URLSearchParams({
    dataInicial: paraPncp(filtro.dataInicial),
    dataFinal: paraPncp(filtro.dataFinal),
    codigoModalidadeContratacao: String(filtro.modalidade),
    uf: filtro.uf,
    pagina: String(pagina),
    tamanhoPagina: String(tamanho),
  });
  return `${BASE_CONSULTA}/v1/contratacoes/${endpoint}?${params}`;
}

export function criarCliente(opcoes: OpcoesCliente = {}) {
  const buscar = opcoes.fetch ?? fetch;
  const esperar = opcoes.esperar ?? ((ms: number) => new Promise<void>((r) => setTimeout(r, ms)));
  const aleatorio = opcoes.aleatorio ?? Math.random;
  const tentativas = opcoes.tentativas ?? TENTATIVAS;
  const timeoutMs = opcoes.timeoutMs ?? TIMEOUT_MS;
  const limitar = criarLimitador(opcoes.paralelismo ?? PARALELISMO);
  const contagem = { chamadas: 0, novasTentativas: 0 };

  async function getJson<T>(url: string): Promise<T | null> {
    return limitar(async () => {
      let ultimoErro: unknown;
      for (let tentativa = 1; tentativa <= tentativas; tentativa++) {
        if (tentativa > 1) {
          contagem.novasTentativas++;
          await esperar(2000 * 2 ** (tentativa - 2) + Math.floor(aleatorio() * 500));
        }
        contagem.chamadas++;
        try {
          const resp = await buscar(url, {
            headers: { Accept: 'application/json' },
            signal: AbortSignal.timeout(timeoutMs),
          });
          if (resp.status === 204) return null;
          if (resp.ok) {
            const corpo = await resp.text();
            return corpo ? (JSON.parse(corpo) as T) : null;
          }
          const erro = new ErroPncp(
            `HTTP ${resp.status} em ${url}: ${(await resp.text()).slice(0, 200)}`,
            resp.status,
            url,
          );
          if (!podeTentarDeNovo(resp.status)) throw erro;
          ultimoErro = erro;
        } catch (erro) {
          if (erro instanceof ErroPncp && erro.status !== undefined && !podeTentarDeNovo(erro.status)) throw erro;
          ultimoErro = erro;
        }
      }
      const detalhe = ultimoErro instanceof Error ? ultimoErro.message : String(ultimoErro);
      throw new ErroPncp(
        `Falhou ${tentativas} vezes: ${url} (${detalhe})`,
        ultimoErro instanceof ErroPncp ? ultimoErro.status : undefined,
        url,
      );
    });
  }

  async function* paginarContratacoes(
    endpoint: 'publicacao' | 'atualizacao',
    filtro: FiltroContratacoes,
  ): AsyncGenerator<PaginaContratacoes> {
    for (let pagina = 1; ; pagina++) {
      const resp = await getJson<RespostaContratacoes>(urlContratacoes(endpoint, filtro, pagina, PAGINA_CONTRATACOES));
      const registros = resp?.data ?? [];
      if (registros.length === 0) return;
      const totalPaginas = resp?.totalPaginas ?? pagina;
      yield { pagina, totalPaginas, totalRegistros: resp?.totalRegistros ?? registros.length, registros };
      if (pagina >= totalPaginas) return;
    }
  }

  async function totalRegistros(endpoint: 'publicacao' | 'atualizacao', filtro: FiltroContratacoes): Promise<number> {
    const resp = await getJson<RespostaContratacoes>(urlContratacoes(endpoint, filtro, 1, 10));
    return resp?.totalRegistros ?? 0;
  }

  async function listarItens(compra: ChaveCompra): Promise<unknown[]> {
    const todos: unknown[] = [];
    for (let pagina = 1; ; pagina++) {
      const url = `${BASE_PNCP}/v1/orgaos/${compra.cnpj}/compras/${compra.ano}/${compra.sequencial}/itens?pagina=${pagina}&tamanhoPagina=${PAGINA_ITENS}`;
      const lote = (await getJson<unknown[]>(url)) ?? [];
      todos.push(...lote);
      if (lote.length < PAGINA_ITENS) return todos;
    }
  }

  async function detalheCompra(compra: ChaveCompra): Promise<unknown | null> {
    return getJson(`${BASE_CONSULTA}/v1/orgaos/${compra.cnpj}/compras/${compra.ano}/${compra.sequencial}`);
  }

  return {
    getJson,
    paginarContratacoes,
    totalRegistros,
    listarItens,
    detalheCompra,
    estatisticas: () => ({ ...contagem }),
  };
}

export type ClientePncp = ReturnType<typeof criarCliente>;
```

import { appendFile } from 'node:fs/promises';

type Ocorrencia = { chave: string; mensagem: string };

export class Resumo {
  readonly inicio: number;
  paginas = 0;
  validos = 0;
  contratacoesGravadas = 0;
  comprasComItens = 0;
  itensGravados = 0;
  readonly invalidos: Array<{ id: string; motivo: string }> = [];
  readonly erros: Ocorrencia[] = [];
  readonly avisos: Ocorrencia[] = [];

  constructor(
    readonly titulo: string,
    private readonly relogio: () => number = Date.now,
  ) {
    this.inicio = relogio();
  }

  contarPagina(): void {
    this.paginas++;
  }
  contarValidos(n: number): void {
    this.validos += n;
  }
  contarInvalido(id: string, motivo: string): void {
    this.invalidos.push({ id, motivo });
  }
  contarContratacoes(n: number): void {
    this.contratacoesGravadas += n;
  }
  contarItens(compras: number, itens: number): void {
    this.comprasComItens += compras;
    this.itensGravados += itens;
  }
  registrarErro(chave: string, mensagem: string): void {
    this.erros.push({ chave, mensagem });
  }
  registrarAviso(chave: string, mensagem: string): void {
    this.avisos.push({ chave, mensagem });
  }

  taxaInvalidos(): number {
    const total = this.validos + this.invalidos.length;
    return total === 0 ? 0 : this.invalidos.length / total;
  }
  falhou(limiteInvalidos: number): boolean {
    return this.erros.length > 0 || this.taxaInvalidos() > limiteInvalidos;
  }
  segundos(): number {
    return Math.round((this.relogio() - this.inicio) / 1000);
  }

  texto(http?: { chamadas: number; novasTentativas: number }): string {
    const pct = (this.taxaInvalidos() * 100).toFixed(1).replace('.', ',');
    const linhas = [
      `## ${this.titulo}`,
      '',
      '| Métrica | Valor |',
      '|---|---|',
      `| Tempo | ${this.segundos()} s |`,
      `| Páginas lidas | ${this.paginas} |`,
      `| Contratações válidas | ${this.validos} |`,
      `| Contratações gravadas | ${this.contratacoesGravadas} |`,
      `| Compras com itens buscados | ${this.comprasComItens} |`,
      `| Itens gravados | ${this.itensGravados} |`,
      `| Registros inválidos | ${this.invalidos.length} (${pct}%) |`,
    ];
    if (http) linhas.push(`| Chamadas HTTP | ${http.chamadas} |`, `| Novas tentativas | ${http.novasTentativas} |`);
    linhas.push(`| Erros | ${this.erros.length} |`, `| Avisos | ${this.avisos.length} |`);

    const secao = (titulo: string, itens: Array<{ rotulo: string; texto: string }>) => {
      if (itens.length === 0) return;
      linhas.push('', `### ${titulo}`, ...itens.slice(0, 20).map((i) => `- \`${i.rotulo}\`: ${i.texto}`));
      if (itens.length > 20) linhas.push(`- … e mais ${itens.length - 20}`);
    };
    secao('Erros', this.erros.map((e) => ({ rotulo: e.chave, texto: e.mensagem })));
    secao('Avisos', this.avisos.map((a) => ({ rotulo: a.chave, texto: a.mensagem })));
    secao('Registros inválidos', this.invalidos.map((i) => ({ rotulo: i.id, texto: i.motivo })));
    return linhas.join('\n');
  }

  async publicar(
    http?: { chamadas: number; novasTentativas: number },
    destino: string | undefined = process.env.GITHUB_STEP_SUMMARY,
  ): Promise<void> {
    const t = this.texto(http);
    console.log(t);
    if (destino) await appendFile(destino, `${t}\n`);
  }
}

import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import { LOTE_BANCO } from './config';
import type { ChaveCompra, LinhaContratacao, LinhaItem } from './mapear';

export type EstadoSync = {
  chave: string;
  ultima_data: string;
  ultima_execucao: string | null;
  status: string | null;
  erro: string | null;
};

export type LinhaMunicipio = { codigo_ibge: string; nome: string; uf: string; lat: number; lng: number };

export interface Banco {
  upsertContratacoes(linhas: LinhaContratacao[]): Promise<void>;
  upsertItens(linhas: LinhaItem[]): Promise<void>;
  removerItensAlem(contratacaoId: string, numeros: number[]): Promise<void>;
  atualizacoesConhecidas(ids: string[]): Promise<Map<string, string>>;
  abertasSemItens(limite: number): Promise<ChaveCompra[]>;
  lerEstado(chave: string): Promise<EstadoSync | null>;
  gravarEstado(estado: EstadoSync): Promise<void>;
}

export type BancoSupabase = Banco & { upsertMunicipios(linhas: LinhaMunicipio[]): Promise<void> };

export function emLotes<T>(itens: T[], tamanho: number): T[][] {
  if (!Number.isInteger(tamanho) || tamanho < 1) throw new Error(`Tamanho de lote inválido: ${tamanho}`);
  const lotes: T[][] = [];
  for (let i = 0; i < itens.length; i += tamanho) lotes.push(itens.slice(i, i + tamanho));
  return lotes;
}

function falhar(contexto: string, erro: { message: string } | null): void {
  if (erro) throw new Error(`${contexto}: ${erro.message}`);
}

export function criarSupabase(url: string, chave: string): SupabaseClient {
  return createClient(url, chave, { auth: { persistSession: false, autoRefreshToken: false } });
}

export function criarBanco(sb: SupabaseClient, agora: () => Date = () => new Date()): BancoSupabase {
  return {
    async upsertContratacoes(linhas) {
      for (const lote of emLotes(linhas, LOTE_BANCO)) {
        const { error } = await sb.from('contratacoes').upsert(lote, { onConflict: 'id' });
        falhar('gravar contratacoes', error);
      }
    },

    async upsertItens(linhas) {
      for (const lote of emLotes(linhas, LOTE_BANCO)) {
        const { error } = await sb.from('itens').upsert(lote, { onConflict: 'contratacao_id,numero' });
        falhar('gravar itens', error);
      }
    },

    async removerItensAlem(contratacaoId, numeros) {
      let consulta = sb.from('itens').delete().eq('contratacao_id', contratacaoId);
      if (numeros.length > 0) consulta = consulta.not('numero', 'in', `(${numeros.join(',')})`);
      const { error } = await consulta;
      falhar('remover itens', error);
    },

    async atualizacoesConhecidas(ids) {
      const mapa = new Map<string, string>();
      for (const lote of emLotes([...new Set(ids)], 200)) {
        const { data, error } = await sb.from('contratacoes').select('id, atualizada_pncp_em').in('id', lote);
        falhar('ler atualizações conhecidas', error);
        for (const linha of data ?? []) mapa.set(linha.id as string, linha.atualizada_pncp_em as string);
      }
      return mapa;
    },

    async abertasSemItens(limite) {
      // Anti-join do PostgREST: embute itens e filtra quem não tem nenhum.
      const { data, error } = await sb
        .from('contratacoes')
        .select('id, cnpj, ano, sequencial, itens(contratacao_id)')
        .gt('encerramento_em', agora().toISOString())
        .is('itens', null)
        .order('encerramento_em', { ascending: true })
        .limit(limite);
      falhar('listar abertas sem itens', error);
      return (data ?? []).map((l) => ({
        id: l.id as string,
        cnpj: l.cnpj as string,
        ano: l.ano as number,
        sequencial: l.sequencial as number,
      }));
    },

    async lerEstado(chave) {
      const { data, error } = await sb
        .from('sync_estado')
        .select('chave, ultima_data, ultima_execucao, status, erro')
        .eq('chave', chave)
        .maybeSingle();
      falhar('ler sync_estado', error);
      return (data as EstadoSync | null) ?? null;
    },

    async gravarEstado(estado) {
      const { error } = await sb.from('sync_estado').upsert(estado, { onConflict: 'chave' });
      falhar('gravar sync_estado', error);
    },

    async upsertMunicipios(linhas) {
      for (const lote of emLotes(linhas, LOTE_BANCO)) {
        const { error } = await sb.from('municipios').upsert(lote, { onConflict: 'codigo_ibge' });
        falhar('gravar municipios', error);
      }
    },
  };
}

import type { Banco } from './banco';
import { LIMITE_CATCH_UP } from './config';
import { hojeBrasilia, somarDias } from './datas';
import {
  chaveDaLinha,
  contratacaoParaLinha,
  estaAberta,
  itemParaLinha,
  type ChaveCompra,
  type LinhaContratacao,
  type LinhaItem,
} from './mapear';
import type { ClientePncp, FiltroContratacoes } from './pncp/cliente';
import { validarContratacao, validarItem } from './pncp/esquemas';
import type { Resumo } from './resumo';

export type Dependencias = {
  cliente: Pick<ClientePncp, 'paginarContratacoes' | 'listarItens'>;
  banco: Banco;
  resumo: Resumo;
  agora: () => Date;
};

export function mensagemDe(erro: unknown): string {
  return erro instanceof Error ? erro.message : String(erro);
}

export function processarRegistros(registros: unknown[], resumo: Resumo): LinhaContratacao[] {
  const porId = new Map<string, LinhaContratacao>();
  for (const bruto of registros) {
    const v = validarContratacao(bruto);
    if (!v.ok) {
      resumo.contarInvalido(v.id, v.motivo);
      continue;
    }
    try {
      const linha = contratacaoParaLinha(v.valor);
      porId.set(linha.id, linha);
    } catch (erro) {
      resumo.contarInvalido(v.valor.numeroControlePNCP, mensagemDe(erro));
    }
  }
  const linhas = [...porId.values()];
  resumo.contarValidos(linhas.length);
  return linhas;
}

export function mudou(linha: LinhaContratacao, conhecidas: Map<string, string>): boolean {
  const anterior = conhecidas.get(linha.id);
  return anterior === undefined || Date.parse(linha.atualizada_pncp_em) > Date.parse(anterior);
}

export async function sincronizarItens(compras: ChaveCompra[], deps: Dependencias): Promise<void> {
  await Promise.all(
    compras.map(async (compra) => {
      try {
        const brutos = await deps.cliente.listarItens(compra);
        const porNumero = new Map<number, LinhaItem>();
        let invalidos = 0;
        for (const bruto of brutos) {
          const v = validarItem(bruto, compra.id);
          if (v.ok) porNumero.set(v.valor.numeroItem, itemParaLinha(v.valor, compra.id));
          else {
            invalidos++;
            deps.resumo.registrarAviso(`item:${v.id}`, v.motivo);
          }
        }
        const linhas = [...porNumero.values()];
        await deps.banco.upsertItens(linhas);
        // Só remove itens antigos quando a lista veio inteira e válida.
        if (invalidos === 0) await deps.banco.removerItensAlem(compra.id, linhas.map((l) => l.numero));
        deps.resumo.contarItens(1, linhas.length);
      } catch (erro) {
        deps.resumo.registrarAviso(`itens:${compra.id}`, mensagemDe(erro));
      }
    }),
  );
}

export async function gravarPagina(linhas: LinhaContratacao[], deps: Dependencias): Promise<void> {
  if (linhas.length === 0) return;
  const conhecidas = await deps.banco.atualizacoesConhecidas(linhas.map((l) => l.id));
  await deps.banco.upsertContratacoes(linhas);
  deps.resumo.contarContratacoes(linhas.length);
  const agora = deps.agora();
  const precisamItens = linhas.filter((l) => estaAberta(l, agora) && mudou(l, conhecidas)).map(chaveDaLinha);
  await sincronizarItens(precisamItens, deps);
}

export async function processarJanela(
  endpoint: 'publicacao' | 'atualizacao',
  filtro: FiltroContratacoes,
  deps: Dependencias,
): Promise<void> {
  for await (const pagina of deps.cliente.paginarContratacoes(endpoint, filtro)) {
    deps.resumo.contarPagina();
    await gravarPagina(processarRegistros(pagina.registros, deps.resumo), deps);
  }
}

export function chaveEstado(uf: string, modalidade: number): string {
  return `atualizacao:${uf}:${modalidade}`;
}

export function inicioDaJanela(ultimaData: string | null | undefined, hoje: string): string {
  const ontem = somarDias(hoje, -1);
  return ultimaData && ultimaData < ontem ? ultimaData : ontem;
}

export async function sincronizarChave(uf: string, modalidade: number, deps: Dependencias): Promise<boolean> {
  const chave = chaveEstado(uf, modalidade);
  const hoje = hojeBrasilia(deps.agora());
  let inicio = somarDias(hoje, -1);
  let estadoLido = false;
  try {
    const estado = await deps.banco.lerEstado(chave);
    estadoLido = true;
    inicio = inicioDaJanela(estado?.ultima_data, hoje);
    await processarJanela('atualizacao', { uf, modalidade, dataInicial: inicio, dataFinal: hoje }, deps);
    await deps.banco.gravarEstado({
      chave,
      ultima_data: hoje,
      ultima_execucao: deps.agora().toISOString(),
      status: 'ok',
      erro: null,
    });
    return true;
  } catch (erro) {
    const mensagem = mensagemDe(erro);
    deps.resumo.registrarErro(chave, mensagem);
    // Sem ter lido o estado, não sabemos a última data boa: não sobrescrever.
    if (estadoLido) {
      await deps.banco
        .gravarEstado({
          chave,
          ultima_data: inicio,
          ultima_execucao: deps.agora().toISOString(),
          status: 'erro',
          erro: mensagem.slice(0, 1000),
        })
        .catch(() => undefined);
    }
    return false;
  }
}

export async function executarSync(deps: Dependencias, ufs: string[], modalidades: readonly number[]): Promise<void> {
  for (const uf of ufs) {
    for (const modalidade of modalidades) await sincronizarChave(uf, modalidade, deps);
  }
  try {
    await sincronizarItens(await deps.banco.abertasSemItens(LIMITE_CATCH_UP), deps);
  } catch (erro) {
    deps.resumo.registrarAviso('catch-up de itens', mensagemDe(erro));
  }
}

import { LIMITE_CATCH_UP } from './config';
import { hojeBrasilia, somarDias } from './datas';
import { chaveEstado, mensagemDe, processarJanela, sincronizarItens, type Dependencias } from './sincronizar';

export function janelas(inicio: string, fim: string, dias: number): Array<[string, string]> {
  const resultado: Array<[string, string]> = [];
  for (let ini = inicio; ini <= fim; ini = somarDias(ini, dias)) {
    const ultimo = somarDias(ini, dias - 1);
    resultado.push([ini, ultimo < fim ? ultimo : fim]);
  }
  return resultado;
}

export async function executarBackfill(
  deps: Dependencias,
  opcoes: { dias: number; ufs: string[]; modalidades: readonly number[] },
): Promise<void> {
  const hoje = hojeBrasilia(deps.agora());
  const inicio = somarDias(hoje, -opcoes.dias);
  for (const uf of opcoes.ufs) {
    for (const modalidade of opcoes.modalidades) {
      const chave = chaveEstado(uf, modalidade);
      try {
        for (const [dataInicial, dataFinal] of janelas(inicio, hoje, 7)) {
          await processarJanela('publicacao', { uf, modalidade, dataInicial, dataFinal }, deps);
        }
        await deps.banco.gravarEstado({
          chave,
          ultima_data: hoje,
          ultima_execucao: deps.agora().toISOString(),
          status: 'ok',
          erro: null,
        });
      } catch (erro) {
        deps.resumo.registrarErro(chave, mensagemDe(erro));
      }
    }
  }
  try {
    await sincronizarItens(await deps.banco.abertasSemItens(LIMITE_CATCH_UP), deps);
  } catch (erro) {
    deps.resumo.registrarAviso('catch-up de itens', mensagemDe(erro));
  }
}

import type { LinhaMunicipio } from './banco';

export const UF_POR_CODIGO: Record<string, string> = {
  '11': 'RO', '12': 'AC', '13': 'AM', '14': 'RR', '15': 'PA', '16': 'AP', '17': 'TO',
  '21': 'MA', '22': 'PI', '23': 'CE', '24': 'RN', '25': 'PB', '26': 'PE', '27': 'AL', '28': 'SE', '29': 'BA',
  '31': 'MG', '32': 'ES', '33': 'RJ', '35': 'SP',
  '41': 'PR', '42': 'SC', '43': 'RS',
  '50': 'MS', '51': 'MT', '52': 'GO', '53': 'DF',
};

export const CABECALHO_MUNICIPIOS = 'codigo_ibge,nome,latitude,longitude,capital,codigo_uf,siafi_id,ddd,fuso_horario';

export function lerMunicipiosCsv(texto: string): LinhaMunicipio[] {
  const linhas = texto
    .replace(/^\uFEFF/, '')
    .split(/\r?\n/)
    .filter((l) => l.trim() !== '');
  if (linhas[0] !== CABECALHO_MUNICIPIOS) throw new Error(`Cabeçalho inesperado: ${linhas[0]}`);
  return linhas.slice(1).map((linha, i) => {
    const campos = linha.split(',');
    if (campos.length !== 9) throw new Error(`Linha ${i + 2}: esperava 9 campos, veio ${campos.length}`);
    const [codigo, nome, latitude, longitude, , codigoUf] = campos;
    const uf = UF_POR_CODIGO[codigoUf];
    if (!uf) throw new Error(`Linha ${i + 2}: codigo_uf desconhecido ${codigoUf}`);
    const lat = Number(latitude);
    const lng = Number(longitude);
    if (!Number.isFinite(lat) || !Number.isFinite(lng)) throw new Error(`Linha ${i + 2}: coordenadas inválidas`);
    return { codigo_ibge: codigo, nome, uf, lat, lng };
  });
}

export function validarMunicipios(linhas: LinhaMunicipio[]): void {
  if (linhas.length !== 5570) throw new Error(`Esperava 5570 municípios, veio ${linhas.length}`);
  const ufs = new Set(linhas.map((m) => m.uf));
  if (ufs.size !== 27) throw new Error(`Esperava 27 UFs, veio ${ufs.size}`);
  const codigos = new Set(linhas.map((m) => m.codigo_ibge));
  if (codigos.size !== linhas.length) throw new Error('Há código IBGE repetido');
}

export type Candidata = {
  id: string;
  cnpj: string;
  ano: number;
  sequencial: number;
  uf: string;
  modalidade_id: number;
  valor_estimado: number | null;
  link_origem: string | null;
  muitos_itens: boolean;
};

export type Sorteada = Candidata & { motivo: string };

export function sortearAmostra(candidatas: Candidata[], tamanho: number, aleatorio: () => number = Math.random): Sorteada[] {
  const escolhidas = new Map<string, Sorteada>();
  const pegar = (filtro: (c: Candidata) => boolean, motivo: string) => {
    if (escolhidas.size >= tamanho) return;
    const opcoes = candidatas.filter((c) => filtro(c) && !escolhidas.has(c.id));
    if (opcoes.length === 0) return;
    const c = opcoes[Math.floor(aleatorio() * opcoes.length)];
    escolhidas.set(c.id, { ...c, motivo });
  };
  const jaTem = (filtro: (c: Sorteada) => boolean) => [...escolhidas.values()].some(filtro);

  pegar((c) => c.muitos_itens, 'mais de 10 itens');
  pegar((c) => c.valor_estimado === 0, 'valor 0');
  pegar((c) => c.link_origem === null, 'link vazio');
  for (const uf of [...new Set(candidatas.map((c) => c.uf))].sort()) {
    if (!jaTem((e) => e.uf === uf)) pegar((c) => c.uf === uf, `UF ${uf}`);
  }
  for (const m of [...new Set(candidatas.map((c) => c.modalidade_id))].sort((a, b) => a - b)) {
    if (!jaTem((e) => e.modalidade_id === m)) pegar((c) => c.modalidade_id === m, `modalidade ${m}`);
  }
  const alvo = Math.min(tamanho, candidatas.length);
  while (escolhidas.size < alvo) pegar(() => true, 'sorteio');
  return [...escolhidas.values()];
}

```ts
import { contratacaoParaLinha, type LinhaContratacao } from '../mapear';
import { validarContratacao } from '../pncp/esquemas';

export type Diferenca = { campo: string; banco: string; pncp: string };
export type ResultadoCompra = {
  id: string;
  status: 'ok' | 'divergente' | 'desatualizada' | 'erro';
  diferencas: Diferenca[];
  observacao?: string;
};

const CAMPOS = [
  'objeto',
  'valor_estimado',
  'encerramento_em',
  'modalidade_id',
  'situacao_id',
  'codigo_ibge',
  'uf',
  'srp',
] as const;

function normalizar(campo: (typeof CAMPOS)[number], valor: unknown): string {
  if (valor === null || valor === undefined) return '∅';
  if (campo === 'encerramento_em') return new Date(valor as string).toISOString();
  if (campo === 'valor_estimado') return String(Number(valor));
  return String(valor).trim();
}

export function compararCompra(
  banco: LinhaContratacao,
  detalheBruto: unknown,
  itensBanco: number,
  itensPncp: number,
): ResultadoCompra {
  const v = validarContratacao(detalheBruto);
  if (!v.ok) return { id: banco.id, status: 'erro', diferencas: [], observacao: `detalhe inválido: ${v.motivo}` };
  const pncp = contratacaoParaLinha(v.valor);
  const diferencas: Diferenca[] = [];
  for (const campo of CAMPOS) {
    const a = normalizar(campo, banco[campo]);
    const b = normalizar(campo, pncp[campo]);
    if (a !== b) diferencas.push({ campo, banco: a, pncp: b });
  }
  if (itensBanco !== itensPncp) diferencas.push({ campo: 'qtd_itens', banco: String(itensBanco), pncp: String(itensPncp) });
  if (Date.parse(pncp.atualizada_pncp_em) > Date.parse(banco.atualizada_pncp_em)) {
    return {
      id: banco.id,
      status: 'desatualizada',
      diferencas,
      observacao: 'O PNCP mudou a compra depois da última sincronização',
    };
  }
  return { id: banco.id, status: diferencas.length > 0 ? 'divergente' : 'ok', diferencas };
}

import { linkDoPortal } from '../mapear';
import type { Sorteada } from './amostra';
import type { ResultadoCompra } from './comparar';

export type LinhaContagem = { dia: string; uf: string; modalidade: number; banco: number; pncp: number };
export type VerificacaoBusca = { verificacao: string; status: 'ok' | 'falhou' | 'inconclusivo'; detalhe: string };
export type Idempotencia = {
  antes: { contratacoes: number; itens: number };
  depois: { contratacoes: number; itens: number };
  novas: number;
  inexplicadas: string[];
};
export type DadosRelatorio = {
  dia: string;
  amostra: Array<{ candidata: Sorteada; resultado: ResultadoCompra }>;
  contagens: LinhaContagem[];
  buscas: VerificacaoBusca[];
  idempotencia: Idempotencia | null;
};

export function veredito(d: DadosRelatorio): { passou: boolean; motivos: string[] } {
  const motivos: string[] = [];
  const ruins = d.amostra.filter((a) => a.resultado.status === 'divergente' || a.resultado.status === 'erro');
  if (ruins.length > 0) motivos.push(`${ruins.length} compra(s) da amostra com divergência ou erro`);
  const contagensErradas = d.contagens.filter((c) => c.banco !== c.pncp);
  if (contagensErradas.length > 0) motivos.push(`${contagensErradas.length} contagem(ns) por dia diferente(s) do PNCP`);
  for (const b of d.buscas) if (b.status !== 'ok') motivos.push(`Busca: ${b.verificacao} (${b.status})`);
  if (!d.idempotencia) motivos.push('Idempotência não verificada (rode com --idempotencia)');
  else if (d.idempotencia.inexplicadas.length > 0) {
    motivos.push(`${d.idempotencia.inexplicadas.length} compra(s) nova(s) na 2ª rodada sem explicação`);
  }
  return { passou: motivos.length === 0, motivos };
}

export function montarRelatorio(d: DadosRelatorio): string {
  const v = veredito(d);
  const linhas: string[] = [`# Conferência · ${d.dia}`, '', '## Veredito', ''];
  linhas.push(v.passou ? '**Passou** nas verificações automáticas.' : '**Não passou.**');
  for (const m of v.motivos) linhas.push(`- ${m}`);

  linhas.push('', `## Amostra (${d.amostra.length} compras)`, '', '| Compra | Por que entrou | Status | Diferenças | Portal |', '|---|---|---|---|---|');
  for (const { candidata, resultado } of d.amostra) {
    const difs = resultado.diferencas.map((x) => `${x.campo}: banco ${x.banco} · PNCP ${x.pncp}`).join('<br>');
    const obs = resultado.observacao ? ` (${resultado.observacao})` : '';
    linhas.push(`| ${candidata.id} | ${candidata.motivo} | ${resultado.status}${obs} | ${difs || '—'} | [abrir](${linkDoPortal(candidata)}) |`);
  }

  const erradas = d.contagens.filter((c) => c.banco !== c.pncp);
  linhas.push('', '## Contagem por dia', '', `${d.contagens.length - erradas.length} de ${d.contagens.length} combinações iguais ao PNCP.`);
  if (erradas.length > 0) {
    linhas.push('', '| Dia | UF | Modalidade | Banco | PNCP |', '|---|---|---|---|---|');
    for (const c of erradas) linhas.push(`| ${c.dia} | ${c.uf} | ${c.modalidade} | ${c.banco} | ${c.pncp} |`);
  }

  linhas.push('', '## Busca', '');
  for (const b of d.buscas) linhas.push(`- ${b.verificacao}: **${b.status}**. ${b.detalhe}`);

  linhas.push('', '## Idempotência', '');
  if (!d.idempotencia) linhas.push('Não verificada nesta execução.');
  else {
    const i = d.idempotencia;
    linhas.push(
      `Contratações ${i.antes.contratacoes} → ${i.depois.contratacoes}; itens ${i.antes.itens} → ${i.depois.itens}.`,
      `Compras novas na 2ª rodada: ${i.novas}; sem explicação: ${i.inexplicadas.length ? i.inexplicadas.join(', ') : 'nenhuma'}.`,
    );
  }

  linhas.push('', '## Conferência manual (dono do produto)', '', 'Abra no portal e confira prazo e valor:', '');
  for (const { candidata } of d.amostra.slice(0, 5)) linhas.push(`- [ ] [${candidata.id}](${linkDoPortal(candidata)})`);
  return `${linhas.join('\n')}\n`;
}

        cnpj: l.cnpj,
        ano: l.ano,
        sequencial: l.sequencial,
        uf: l.uf,
        modalidade_id: l.modalidade_id,
        valor_estimado: l.valor_estimado === null ? null : Number(l.valor_estimado),
        link_origem: l.link_origem,
        muitos_itens: muitos.has(l.id),
      });
    }
    if (!data || data.length < 1000) return candidatas;
  }
}

async function conferirAmostra() {
  const amostra = sortearAmostra(await listarCandidatas(), tamanho);
  const resultados = [];
  for (const candidata of amostra) {
    let resultado: ResultadoCompra;
    try {
      const linha = await sb.from('contratacoes').select('*').eq('id', candidata.id).single();
      if (linha.error) throw new Error(linha.error.message);
      const itens = await sb.from('itens').select('numero', { count: 'exact', head: true }).eq('contratacao_id', candidata.id);
      if (itens.error) throw new Error(itens.error.message);
      const [detalhe, itensPncp] = await Promise.all([cliente.detalheCompra(candidata), cliente.listarItens(candidata)]);
      resultado = compararCompra(linha.data as LinhaContratacao, detalhe, itens.count ?? 0, itensPncp.length);
    } catch (erro) {
      resultado = { id: candidata.id, status: 'erro', diferencas: [], observacao: mensagemDe(erro) };
    }
    resultados.push({ candidata, resultado });
  }
  return resultados;
}

function sortearDias(n: number): string[] {
  const dias = new Set<string>();
  while (dias.size < n) dias.add(somarDias(hoje, -(1 + Math.floor(Math.random() * 89))));
  return [...dias].sort();
}

async function contarDias(): Promise<LinhaContagem[]> {
  const linhas: LinhaContagem[] = [];
  for (const dia of sortearDias(diasContagem)) {
    for (const uf of ufs) {
      for (const modalidade of modalidades) {
        const { count, error } = await sb
          .from('contratacoes')
          .select('id', { count: 'exact', head: true })
          .eq('uf', uf)
          .eq('modalidade_id', modalidade)
          .gte('publicada_em', inicioDoDia(dia))
          .lt('publicada_em', inicioDoDia(somarDias(dia, 1)));
        if (error) throw new Error(error.message);
        const pncp = await cliente.totalRegistros('publicacao', { uf, modalidade, dataInicial: dia, dataFinal: dia });
        linhas.push({ dia, uf, modalidade, banco: count ?? 0, pncp });
      }
    }
  }
  return linhas;
}

async function buscar(termo: string): Promise<Array<{ id: string; objeto: string }>> {
  const { data, error } = await sb.rpc('buscar', { q: termo, p_limite: 1000 });
  if (error) throw new Error(error.message);
  return (data ?? []) as Array<{ id: string; objeto: string }>;
}

async function conferirBuscas(): Promise<VerificacaoBusca[]> {
  const resultado: VerificacaoBusca[] = [];
  let achado: { termo: string; id: string } | null = null;
  for (const termo of ['botina', 'luva', 'capacete']) {
    const so = (await buscar(termo)).find((x) => !semAcento(x.objeto).toLowerCase().includes(termo));
    if (so) {
      achado = { termo, id: so.id };
      break;
    }
  }
  resultado.push(
    achado
      ? { verificacao: 'Busca acha pelos itens', status: 'ok', detalhe: `"${achado.termo}" achou ${achado.id}, cujo objeto não tem a palavra.` }
      : { verificacao: 'Busca acha pelos itens', status: 'inconclusivo', detalhe: 'Nenhuma compra aberta com botina, luva ou capacete só nos itens.' },
  );
  const [epi, uniformes, ambos] = await Promise.all([buscar('EPI'), buscar('uniformes'), buscar('EPI e uniformes')]);
  const ids = new Set(ambos.map((x) => x.id));
  const traz = epi.some((x) => ids.has(x.id)) && uniformes.some((x) => ids.has(x.id));
  resultado.push({
    verificacao: 'Busca "EPI e uniformes" traz os dois',
    status: epi.length === 0 || uniformes.length === 0 ? 'inconclusivo' : traz ? 'ok' : 'falhou',
    detalhe: `EPI: ${epi.length}; uniformes: ${uniformes.length}; os dois: ${ambos.length}.`,
  });
  return resultado;
}

async function contarTabelas() {
  const c = await sb.from('contratacoes').select('id', { count: 'exact', head: true });
  const i = await sb.from('itens').select('numero', { count: 'exact', head: true });
  if (c.error || i.error) throw new Error((c.error ?? i.error)?.message);
  return { contratacoes: c.count ?? 0, itens: i.count ?? 0 };
}

async function conferirIdempotencia(): Promise<Idempotencia> {
  const banco = criarBanco(sb);
  const rodar = async (titulo: string) => {
    const resumo = new Resumo(titulo);
    await executarSync({ cliente, banco, resumo, agora: () => new Date() }, ufs, modalidades);
    await resumo.publicar(cliente.estatisticas(), undefined);
  };
  const t0 = Date.now();
  await rodar('Idempotência · 1ª rodada');
  const t1 = new Date();
  const antes = await contarTabelas();
  await rodar('Idempotência · 2ª rodada');
  const depois = await contarTabelas();
  // sincronizada_em só é preenchida na inserção (o upsert não envia essa coluna).
  const { data, error } = await sb.from('contratacoes').select('id, atualizada_pncp_em').gte('sincronizada_em', t1.toISOString());
  if (error) throw new Error(error.message);
  const folga = 10 * 60 * 1000;
  const inexplicadas = (data ?? [])
    .filter((l) => Date.parse(l.atualizada_pncp_em as string) < t0 - folga)
    .map((l) => l.id as string);
  return { antes, depois, novas: data?.length ?? 0, inexplicadas };
}

const relatorio = montarRelatorio({
  dia: hoje,
  amostra: await conferirAmostra(),
  contagens: await contarDias(),
  buscas: await conferirBuscas(),
  idempotencia: values.idempotencia ? await conferirIdempotencia() : null,
});
await mkdir('docs/conferencias', { recursive: true });
const arquivo = `docs/conferencias/${hoje}.md`;
await writeFile(arquivo, relatorio);
console.log(relatorio);
console.log(`Relatório salvo em ${arquivo}`);
```

-- Texto do usuário -> tsquery por prefixo ("epi" acha "EPIs").
-- Vírgula, ";", " e " e " ou " separam alternativas: "EPI e uniformes" = EPI OU uniformes.
create or replace function public.consulta(q text) returns tsquery
language plpgsql immutable as $$
declare
  grupos text[];
  g text;
  partes text[];
  tq tsquery;
  resultado tsquery;
begin
  if coalesce(trim(q), '') = '' then return null; end if;
  grupos := regexp_split_to_array(lower(public.sem_acento(q)), '\s*[,;|]\s*|\s+(e|ou)\s+');
  foreach g in array grupos loop
    select array_agg(w || ':*') into partes
      from regexp_split_to_table(g, '[^a-z0-9]+') w
     where length(w) >= 2;
    continue when partes is null;
    tq := to_tsquery('portuguese', array_to_string(partes, ' & '));
    continue when numnode(tq) = 0;
    resultado := case when resultado is null then tq else resultado || tq end;
  end loop;
  return resultado;
end $$;

-- Oportunidades abertas que casam com a busca (base da lista e do mapa)
create or replace function public.abertas(q text, p_ufs text[] default null, p_valor_min numeric default null)
returns setof public.contratacoes language sql stable as $$
  with t as (select public.consulta(q) as tq)
  select c.*
  from public.contratacoes c, t
  where c.encerramento_em > now()
    and (t.tq is null
         or c.busca @@ t.tq
         or exists (select 1 from public.itens i where i.contratacao_id = c.id and i.busca @@ t.tq))
    and (p_ufs is null or cardinality(p_ufs) = 0 or c.uf = any(p_ufs))
    and (p_valor_min is null or c.valor_estimado >= p_valor_min)
$$;

-- Lista: prazo mais próximo primeiro (o servidor aplica o limite do plano)
create or replace function public.buscar(q text, p_ufs text[] default null, p_valor_min numeric default null,
                                         p_limite int default 10, p_offset int default 0)
returns table (id text, slug text, objeto text, orgao_nome text, municipio_nome text, uf text,
               valor_estimado numeric, encerramento_em timestamptz, modalidade_nome text)
language sql stable as $$
  select a.id, a.slug, a.objeto, a.orgao_nome, a.municipio_nome, a.uf,
         a.valor_estimado, a.encerramento_em, a.modalidade_nome
  from public.abertas(q, p_ufs, p_valor_min) a
  order by a.encerramento_em asc
  limit p_limite offset p_offset
$$;

-- Mapa: um ponto por município, com contagem, valor e prazo mais próximo
create or replace function public.mapa(q text, p_ufs text[] default null, p_valor_min numeric default null)
returns table (codigo_ibge text, nome text, uf text, lat double precision, lng double precision,
               qtd bigint, valor_total numeric, prazo_mais_proximo timestamptz)
language sql stable as $$
  select m.codigo_ibge, m.nome, m.uf, m.lat, m.lng,
         count(*), sum(nullif(a.valor_estimado, 0)), min(a.encerramento_em)
  from public.abertas(q, p_ufs, p_valor_min) a
  join public.municipios m on m.codigo_ibge = a.codigo_ibge
  group by m.codigo_ibge, m.nome, m.uf, m.lat, m.lng
$$;
